% --- Compact title block command ---------------------------------------------
% Renders title, optional subtitle, and date at the top of page 1.
% Usage (called automatically via include-before-body):
%   \brieftitle{Title}{Subtitle}{Date}
\newcommand{\brieftitle}[3]{%
  \vspace*{-0.5em}%
  {\fontsize{22}{26}\selectfont\bfseries #1\par}%
  \ifx&#2&\else%
    \vspace{0.3em}%
    {\large\textcolor{subtlegray}{#2}\par}%
  \fi%
  \vspace{0.3em}%
  {\normalsize\textcolor{midgray}{#3}\par}%
  \vspace{0.2em}%
  {\color{lightgray}\rule{\textwidth}{0.4pt}}%
  \vspace{0.8em}%
}
